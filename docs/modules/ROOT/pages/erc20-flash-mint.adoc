= ERC-20 Flash Mint

Extension of xref:erc20.adoc[ERC-20] that provides flash loan support at the token level.

The flash loan callback must be able to call the lender again to approve its
repayment. With Stylus SDK 0.10.10, enable the `reentrant` feature on the
library dependency so the entrypoint permits that callback flow:

[source,toml]
----
[dependencies]
openzeppelin-stylus = { workspace = true, features = ["reentrant"] }
----

This opt-in permits all reentrant entrypoint calls. Keep state changes before
external callbacks and apply any additional checks your contract requires.

[[usage]]
== Usage

In order to make https://docs.rs/openzeppelin-stylus/0.3.0/openzeppelin_stylus/token/erc20/extensions/flash_mint/index.html[`ERC-20 Flash Mint`]  methods “external” so that other contracts can call them, you need to add the following code to your contract:

[source,rust]
----
use openzeppelin_stylus::token::erc20::{
    extensions::{flash_mint, Erc20FlashMint, IErc3156FlashLender},
    Erc20, IErc20,
};

#[entrypoint]
#[storage]
struct Erc20FlashMintExample {
    erc20: Erc20,
    flash_mint: Erc20FlashMint,
}

#[public]
#[implements(IErc20<Error = flash_mint::Error>, IErc3156FlashLender<Error = flash_mint::Error>)]
impl Erc20FlashMintExample {}

#[public]
impl IErc3156FlashLender for Erc20FlashMintExample {
    type Error = flash_mint::Error;

    fn max_flash_loan(&self, token: Address) -> U256 {
        self.flash_mint.max_flash_loan(token, &self.erc20)
    }

    fn flash_fee(
        &self,
        token: Address,
        value: U256,
    ) -> Result<U256, flash_mint::Error> {
        self.flash_mint.flash_fee(token, value)
    }

    fn flash_loan(
        &mut self,
        receiver: Address,
        token: Address,
        value: U256,
        data: Bytes,
    ) -> Result<bool, flash_mint::Error> {
        self.flash_mint.flash_loan(
            receiver,
            token,
            value,
            &data,
            &mut self.erc20,
        )
    }
}

#[public]
impl IErc20 for Erc20FlashMintExample {
    type Error = flash_mint::Error;

    fn total_supply(&self) -> U256 {
        self.erc20.total_supply()
    }

    fn balance_of(&self, account: Address) -> U256 {
        self.erc20.balance_of(account)
    }

    fn transfer(
        &mut self,
        to: Address,
        value: U256,
    ) -> Result<bool, flash_mint::Error> {
        Ok(self.erc20.transfer(to, value)?)
    }

    fn allowance(&self, owner: Address, spender: Address) -> U256 {
        self.erc20.allowance(owner, spender)
    }

    fn approve(
        &mut self,
        spender: Address,
        value: U256,
    ) -> Result<bool, flash_mint::Error> {
        Ok(self.erc20.approve(spender, value)?)
    }

    fn transfer_from(
        &mut self,
        from: Address,
        to: Address,
        value: U256,
    ) -> Result<bool, flash_mint::Error> {
        Ok(self.erc20.transfer_from(from, to, value)?)
    }
}
----
